自动创建 school_settings、pajsk_records 数据表、RLS 策略及 pajsk-pdfs 存储桶 

-- 1. 创建学校资料与校徽配置表 (public.school_settings)
CREATE TABLE IF NOT EXISTS public.school_settings (
    id INTEGER PRIMARY KEY DEFAULT 1 CHECK (id = 1),
    school_name_zh TEXT NOT NULL DEFAULT '中华小学',
    school_name_bm TEXT NOT NULL DEFAULT 'SEKOLAH JENIS KEBANGSAAN (CINA) CHUNG HUA RANTAU',
    kod_sekolah TEXT NOT NULL DEFAULT 'NBC4067',
    sesi_akademik TEXT NOT NULL DEFAULT '2025/2026',
    logo_url TEXT DEFAULT '',
    logo_path TEXT DEFAULT '',
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

INSERT INTO public.school_settings (id, school_name_zh, school_name_bm, kod_sekolah, sesi_akademik)
VALUES (1, '中华小学', 'SEKOLAH JENIS KEBANGSAAN (CINA) CHUNG HUA RANTAU', 'NBC4067', '2025/2026')
ON CONFLICT (id) DO NOTHING;

ALTER TABLE public.school_settings ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public can read school settings" ON public.school_settings;
CREATE POLICY "Public can read school settings"
    ON public.school_settings FOR SELECT
    USING (true);

DROP POLICY IF EXISTS "Authenticated users can insert school settings" ON public.school_settings;
CREATE POLICY "Authenticated users can insert school settings"
    ON public.school_settings FOR INSERT
    TO authenticated
    WITH CHECK (true);

DROP POLICY IF EXISTS "Authenticated users can update school settings" ON public.school_settings;
CREATE POLICY "Authenticated users can update school settings"
    ON public.school_settings FOR UPDATE
    TO authenticated
    USING (true)
    WITH CHECK (true);

-- 2. 创建 PAJSK 成绩主表 (public.pajsk_records)
CREATE TABLE IF NOT EXISTS public.pajsk_records (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    mykad TEXT UNIQUE NOT NULL,
    nama TEXT NOT NULL,
    kelas TEXT,
    tahun TEXT,
    sesi TEXT DEFAULT '2025/2026',
    kod_sekolah TEXT DEFAULT 'NBC4067',
    sekolah TEXT DEFAULT 'SEKOLAH JENIS KEBANGSAAN (CINA) CHUNG HUA RANTAU',
    sukan_nama TEXT,
    sukan_skor INTEGER DEFAULT 0,
    sukan_peratus NUMERIC(5,2) DEFAULT 0,
    kelab_nama TEXT,
    kelab_skor INTEGER DEFAULT 0,
    kelab_peratus NUMERIC(5,2) DEFAULT 0,
    uniform_nama TEXT,
    uniform_skor INTEGER DEFAULT 0,
    uniform_peratus NUMERIC(5,2) DEFAULT 0,
    ekstra_skor NUMERIC(5,2) DEFAULT 0,
    gpa NUMERIC(5,2) DEFAULT 0,
    markah_10_peratus NUMERIC(4,2) DEFAULT 0,
    gred TEXT,
    pernyataan TEXT,
    pdf_path TEXT,
    pdf_url TEXT,
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_pajsk_records_mykad ON public.pajsk_records(mykad);

ALTER TABLE public.pajsk_records ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public can read pajsk records" ON public.pajsk_records;
CREATE POLICY "Public can read pajsk records"
    ON public.pajsk_records FOR SELECT
    USING (true);

DROP POLICY IF EXISTS "Authenticated users can insert pajsk records" ON public.pajsk_records;
CREATE POLICY "Authenticated users can insert pajsk records"
    ON public.pajsk_records FOR INSERT
    TO authenticated
    WITH CHECK (true);

DROP POLICY IF EXISTS "Authenticated users can update pajsk records" ON public.pajsk_records;
CREATE POLICY "Authenticated users can update pajsk records"
    ON public.pajsk_records FOR UPDATE
    TO authenticated
    USING (true)
    WITH CHECK (true);

DROP POLICY IF EXISTS "Authenticated users can delete pajsk records" ON public.pajsk_records;
CREATE POLICY "Authenticated users can delete pajsk records"
    ON public.pajsk_records FOR DELETE
    TO authenticated
    USING (true);

-- 3. 创建公开存储桶 pajsk-pdfs 存放官方原版 PDF 与校徽图片
INSERT INTO storage.buckets (id, name, public)
VALUES ('pajsk-pdfs', 'pajsk-pdfs', true)
ON CONFLICT (id) DO UPDATE SET public = true;

DROP POLICY IF EXISTS "Public read access for pajsk-pdfs" ON storage.objects;
CREATE POLICY "Public read access for pajsk-pdfs"
    ON storage.objects FOR SELECT
    USING (bucket_id = 'pajsk-pdfs');

DROP POLICY IF EXISTS "Authenticated upload for pajsk-pdfs" ON storage.objects;
CREATE POLICY "Authenticated upload for pajsk-pdfs"
    ON storage.objects FOR INSERT
    TO authenticated
    WITH CHECK (bucket_id = 'pajsk-pdfs');

DROP POLICY IF EXISTS "Authenticated update for pajsk-pdfs" ON storage.objects;
CREATE POLICY "Authenticated update for pajsk-pdfs"
    ON storage.objects FOR UPDATE
    TO authenticated
    USING (bucket_id = 'pajsk-pdfs');

DROP POLICY IF EXISTS "Authenticated delete for pajsk-pdfs" ON storage.objects;
CREATE POLICY "Authenticated delete for pajsk-pdfs"
    ON storage.objects FOR DELETE
    TO authenticated
    USING (bucket_id = 'pajsk-pdfs');
